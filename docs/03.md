  href,

        {description}

  );
}

# 2. 全站加载 Navbar

这是最关键的一步。

打开：

```text
src/app/layout.tsx
```

改成：

```tsx
import type { Metadata } from "next";
import Navbar from "@/components/Navbar";
import "./globals.css";

export const metadata: Metadata = {
  metadataBase: new URL(
    "https://sellercalc-ai.vercel.app"
  ),

  title: {
    default: "SellerCalc AI - TikTok Seller Tools",
    template: "%s | SellerCalc AI",
  },

  description:
    "Free TikTok Shop seller calculators for profit, fees and ROAS.",

  robots: {
    index: true,
    follow: true,
  },
};

export default function RootLayout({
  children,
}: Readonly<{
  children: React.ReactNode;
}>) {
  return (
    <html lang="en">

      <body className="bg-gray-50 text-gray-900">

        <Navbar />

        {children}

      </body>

    </html>
  );
}
```

这样：

```text
/
 /tools
 /profit-calculator
 /fee-calculator
 /roas-calculator
 /blog
```

**全部自动拥有同一个 Navbar。**
